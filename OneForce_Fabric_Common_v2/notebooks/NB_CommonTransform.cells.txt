import json
try:
    validate_execution_id(execution_id)
    log_event("INFO", "COMMON_START", execution_id)
    result = run_transform_job(spark, param_json, execution_id)
    log_event("INFO", "COMMON_END", execution_id, row_count=result["row_count"],
              changed_count=result["changed_count"], write_output=result["write_output"])
    result_json = json.dumps(result, ensure_ascii=False, allow_nan=False)
except Exception as exc:
    # 原文や例外詳細を通常ログに出さない。Fabricの例外証跡の閲覧権限も管理する。
    safe_id = execution_id if isinstance(execution_id, str) and len(execution_id) <= 64 else "invalid"
    log_event("ERROR", "COMMON_FAILED", safe_id, error_type=type(exc).__name__,
              error_code=getattr(exc, "code", "FABRIC_RUNTIME"))
    raise
